        // Get Geolocation Coordinates
        function getCurrentLocation(textId, latId, lngId) {
            const btnText = document.getElementById(textId);
            btnText.innerText = "กำลังค้นหาพิกัด GPS...";

            if (navigator.geolocation) {
                navigator.geolocation.getCurrentPosition(
                    (position) => {
                        const lat = position.coords.latitude;
                        const lng = position.coords.longitude;
                        document.getElementById(latId).value = lat;
                        document.getElementById(lngId).value = lng;
                        btnText.innerText = `ดึงพิกัดสำเร็จ: ${lat.toFixed(4)}, ${lng.toFixed(4)} ✓`;
                        btnText.parentElement.className = "w-full bg-emerald-50 border border-emerald-300 text-emerald-800 py-2 px-3 rounded-xl flex items-center justify-center gap-1.5 font-bold transition";
                    },
                    (error) => {
                        alert("ไม่สามารถเข้าถึงตำแหน่งพิกัด GPS ได้ กรุณาเปิดระบบระบุตำแหน่งของอุปกรณ์");
                        btnText.innerText = "กดเพื่อระบุพิกัด GPS เรียลไทม์";
                    }
                );
            } else {
                alert("เบราว์เซอร์ของคุณไม่รองรับการดึงพิกัด GPS");
                btnText.innerText = "กดเพื่อระบุพิกัด GPS เรียลไทม์";
            }
        }

        // Handle SOS Submit
        function handleSosSubmit(e) {
            e.preventDefault();
            const name = document.getElementById('sos-name').value;
            const phone = document.getElementById('sos-phone').value;
            const zone = document.getElementById('sos-address').value;
            const detailLoc = document.getElementById('sos-detail-loc').value;
            const type = document.getElementById('sos-type').value;
            const lat = document.getElementById('sos-lat').value;
            const lng = document.getElementById('sos-lng').value;

            let gpsString = "ไม่ได้แนบพิกัด GPS";
            let googleMapsUrl = "";
            if (lat && lng) {
                gpsString = `${lat}, ${lng}`;
                googleMapsUrl = `\n📍 แผนที่: https://maps.google.com/?q=${lat},${lng}`;
            }

            const messageText = `🚨 *แจ้งขอความช่วยเหลือฉุกเฉิน SOS (บ้านคลอแระ)*\n👤 ชื่อ: ${name}\n📞 เบอร์โทร: ${phone}\n📍 โซน: ${zone} (${detailLoc})\n🆘 ต้องการ: ${type}\n🌐 พิกัด GPS: ${gpsString}${googleMapsUrl}`;

            // Copy SOS message to clipboard
            if (navigator.clipboard && window.isSecureContext) {
                navigator.clipboard.writeText(messageText);
            } else {
                const tempInput = document.createElement("textarea");
                tempInput.value = messageText;
                document.body.appendChild(tempInput);
                tempInput.select();
                document.execCommand("copy");
                document.body.removeChild(tempInput);
            }

            alert("คัดลอกข้อความ SOS พร้อมพิกัด GPS เรียบร้อยแล้ว!\nกำลังเปิด Messenger ทีมกู้ภัย...");
            closeSosModal();

            // Direct Redirect to Rescue Messenger
            window.open(rescueMessengerLink, '_blank');
        }

        // Handle Evacuee Registration Submit
        function handleEvacueeSubmit(e) {
            e.preventDefault();
            const shelter = document.getElementById('evac-shelter').value;
            const name = document.getElementById('evac-name').value;
            const age = document.getElementById('evac-age').value || '-';
            const group = document.getElementById('evac-group').value;
            const zone = document.getElementById('evac-zone').value;
            const room = document.getElementById('evac-room').value || 'ไม่ระบุ';
            const phone = document.getElementById('evac-phone').value;

            const list = document.getElementById('evacuees-list');
            
            let groupBadgeClass = "bg-blue-100 text-blue-800";
            if (group === "ผู้สูงอายุ") groupBadgeClass = "bg-purple-100 text-purple-800";
            else if (group === "ผู้ป่วย/ผู้พิการ") groupBadgeClass = "bg-red-100 text-red-800";
            else if (group === "หญิงตั้งครรภ์") groupBadgeClass = "bg-pink-100 text-pink-800";

            const newCard = document.createElement('div');
            newCard.className = "p-2.5 bg-slate-50 border border-slate-200 rounded-xl text-xs flex justify-between items-center evacuee-item animate-fade-in";
            newCard.innerHTML = `
                <div>
                    <div class="flex items-center gap-1.5">
                        <span class="font-bold text-slate-800">${name} (${age} ปี)</span>
                        <span class="${groupBadgeClass} text-[9px] font-bold px-1.5 py-0.2 rounded">${group}</span>
                    </div>
                    <p class="text-[10px] text-slate-500">มาจาก: ${zone} | พักที่: ${shelter}</p>
                    <p class="text-[10px] text-emerald-700 font-semibold">ติดต่อ: ${phone}</p>
                </div>
                <span class="text-[9px] bg-slate-200 text-slate-700 px-2 py-1 rounded-md font-bold whitespace-nowrap">${room}</span>
            `;

            list.insertBefore(newCard, list.firstChild);

            // Update Count
            const itemsCount = list.querySelectorAll('.evacuee-item').length;
            document.getElementById('evacuee-count-badge').innerText = `${itemsCount} คน`;

            alert("บันทึกข้อมูลผู้อพยพเรียบร้อยแล้ว!");
            closeEvacueeModal();
        }

        // Filter/Search Evacuees List
        function filterEvacuees() {
            const query = document.getElementById('evacuee-search').value.toLowerCase();
            const items = document.querySelectorAll('.evacuee-item');

            items.forEach(item => {
                const text = item.innerText.toLowerCase();
                if (text.includes(query)) {
                    item.style.display = "flex";
                } else {
                    item.style.display = "none";
                }
            });
        }
